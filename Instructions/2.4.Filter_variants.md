# Filtering variants

# mac

# GQ and DP > 2stds

Example script:

```bash
#!/bin/bash

# Usage: ./filter_parallel.sh input.vcf NUM_THREADS
set -e

if [[ $# -ne 2 ]]; then
    echo "Usage: $0 input.vcf NUM_THREADS"
    exit 1
fi

input_vcf="$1"
num_threads="$2"

mkdir -p filtered_samples

# Step 1: Get sample names
bcftools query -l "$input_vcf" > samples.txt

# Step 2: Define processing function
process_sample() {
    sample="$1"
    echo "Processing $sample"

    # Extract sample VCF
    bcftools view -s "$sample" -Ov -o "${sample}.vcf" "$input_vcf"

    # Extract non-missing DP values
    dp_values=$(bcftools query -f '[%DP\n]' "${sample}.vcf" | grep -v '\.')

    # Check for empty
    if [[ -z "$dp_values" ]]; then
        echo "No DP values for $sample, skipping"
        return
    fi

    # Calculate mean and SD
    mean=$(echo "$dp_values" | awk '{sum+=$1} END {if(NR>0) print sum/NR}')
    sd=$(echo "$dp_values" | awk -v mean="$mean" '{sum+=($1 - mean)^2} END {if(NR>1) print sqrt(sum/(NR-1)); else print 0}')

    # Bounds
    upper=$(awk -v m="$mean" -v s="$sd" 'BEGIN { print m + 2 * s }')

    echo "$sample - DP between $lower and $upper"

    # Filter and write to output dir
    bcftools view -i "FORMAT/DP<=$upper && FORMAT/GQ>=30" -Ov -o "filtered_samples/${sample}.vcf" "${sample}.vcf"

    # Compress and index
    bgzip "filtered_samples/${sample}.vcf"
    tabix "filtered_samples/${sample}.vcf.gz"
}

export -f process_sample
export input_vcf

# Step 3: Run in parallel using user-defined thread count
cat samples.txt | parallel -j "$num_threads" process_sample

# Step 4: Merge all filtered samples
bcftools merge filtered_samples/*.vcf.gz -Ov -o merged_filtered.vcf
echo "Output written to merged_filtered.vcf"
```
