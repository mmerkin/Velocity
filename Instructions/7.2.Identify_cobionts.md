# Identification of cobionts with phyloflash

```bash
datapath=~/Velocity/E3_Aphantopus_hyperantus/reads/modc
for i in $datapath/*R1*; do filetag=$(basename $i .veladapt.clean_R1.fastq.gz); phyloFlash.pl -sortmerna -skip_spades -lib $filetag -CPUs 16 -dbhome ~/apps/phyloflash/SILVA/138.2 -read1 $datapath/${filetag}.veladapt.clean_R1.fastq.gz -read2 $datapath/${filetag}.veladapt.clean_R2.fastq.gz; done
```
