# Filtering variants


Initially, variants are just called using the modern samples with bcftools:

```bash
#!/bin/bash
# Usage: bash script.sh base reference_path bed_path
# Requirements: raw vcf, rename file

# Variables to change:
threads=32
remove_temp=false

base=$1
REF=$2
BED=$3

#bcftools mpileup -b ${base}_bamlist.txt -f $REF -a "FORMAT/QS,FORMAT/AD,FORMAT/DP,INFO/AD" -B | bcftools call -m --threads $threads --ploidy 2 -a GQ,GP -v -Oz -o ${base}_variants.raw.vcf.gz
echo "SNPs called" > ${base}_SNP_count.txt
bcftools view -H ${base}_variants.raw.vcf.gz | wc -l >> ${base}_SNP_count.txt

bcftools filter -g 5 -i'QUAL >= 30' -Ou ${base}_variants.raw.vcf.gz | \
bcftools view -m2 -M2 -v snps -Ou | \
bcftools +fill-tags -Ou -- -t all | \
bcftools filter -e 'F_MISSING>0.5' -Oz > ${base}_variants.filter1.vcf.gz
tabix ${base}_variants.filter1.vcf.gz
echo "SNPs after basic filters" >> ${base}_SNP_count.txt
bcftools view -H ${base}_variants.filter1.vcf.gz | wc -l >> ${base}_SNP_count.txt

# Filter out individuals

vcftools --gzvcf renamed_${base}_variants.filter1.vcf.gz --missing-indv
cat out.imiss | sort -nk 5
mawk '$5 > 0.2' out.imiss | cut -f1 > lowDP.indv
vcftools --gzvcf ${base}_variants.filter1.vcf.gz --remove lowDP.indv --recode --recode-INFO-all --out ${base}_variants.filter1.5
bgzip ${base}_variants.filter1.5.recode.vcf
tabix ${base}_variants.filter1.5.recode.vcf.gz

#  Depth filters

stats=$(bcftools query -f '%INFO/DP\n' ${base}_variants.filter1.5.recode.vcf.gz | \
awk '$1>0 {
  n++
  sum+=$1
  sumsq+=$1*$1

  # reservoir-like tracking for median approximation
  if(n==1){min=max=$1}
  if($1<min) min=$1
  if($1>max) max=$1
}
END {
  mean=sum/n
  sd=sqrt(sumsq/n - mean*mean)

  # crude median approximation (very stable for DP distributions)
  median=(min+max)/2

  print median, sd
}')


read median sd <<< "$stats"
cutoff=$(awk -v m="$median" -v s="$sd" 'BEGIN {print m + 2*s}')
echo $cutoff

bcftools filter \
  -e "INFO/DP < 3 || INFO/DP > $cutoff" \
  -Oz -o ${base}_filter2.vcf.gz \
  ${base}_variants.filter1.5.recode.vcf.gz


```

Example script:

```bash
#!/bin/bash

# Usage: ./filter_parallel.sh input.vcf NUM_THREADS
set -e

if [[ $# -ne 2 ]]; then
    echo "Usage: $0 input.vcf NUM_THREADS"
    exit 1
fi

input_vcf="$1"
num_threads="$2"

mkdir -p filtered_samples

# Step 1: Get sample names
bcftools query -l "$input_vcf" > samples.txt

# Step 2: Define processing function
process_sample() {
    sample="$1"
    echo "Processing $sample"

    # Extract sample VCF
    bcftools view -s "$sample" -Ov -o "${sample}.vcf" "$input_vcf"

    # Extract non-missing DP values
    dp_values=$(bcftools query -f '[%DP\n]' "${sample}.vcf" | grep -v '\.')

    # Check for empty
    if [[ -z "$dp_values" ]]; then
        echo "No DP values for $sample, skipping"
        return
    fi

    # Calculate mean and SD
    mean=$(echo "$dp_values" | awk '{sum+=$1} END {if(NR>0) print sum/NR}')
    sd=$(echo "$dp_values" | awk -v mean="$mean" '{sum+=($1 - mean)^2} END {if(NR>1) print sqrt(sum/(NR-1)); else print 0}')

    # Bounds
    upper=$(awk -v m="$mean" -v s="$sd" 'BEGIN { print m + 2 * s }')

    echo "$sample - DP between $lower and $upper"

    # Filter and write to output dir
    bcftools view -i "FORMAT/DP<=$upper && FORMAT/GQ>=10" -Ov -o "filtered_samples/${sample}.vcf" "${sample}.vcf"

    # Compress and index
    bgzip "filtered_samples/${sample}.vcf"
    tabix "filtered_samples/${sample}.vcf.gz"
}

export -f process_sample
export input_vcf

# Step 3: Run in parallel using user-defined thread count
cat samples.txt | parallel -j "$num_threads" process_sample

# Step 4: Merge all filtered samples
bcftools merge filtered_samples/*.vcf.gz -Ov -o merged_filtered.vcf
echo "Output written to merged_filtered.vcf"
```
