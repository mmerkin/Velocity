Tally up

```bash
for i in */; do 
mod_pi=$(cat $i/output/pixy_pi.txt | awk '$1 == "modern" { comps+=$7; diffs+=$8 } END {print comps/diffs}');
mus_pi=$(cat $i/output/pixy_pi.txt | awk '$1 == "museum" { comps+=$7; diffs+=$8 } END {print comps/diffs}');
fold_change=$(echo "$mus_pi / $mod_pi" | bc -l);
echo -e "${i}\t${mod_pi}\t${mus_pi}\t${fold_change}"; 
done

for i in */; do 
mod_theta=$(cat $i/output/pixy_watterson_theta.txt | awk '$1 == "modern" { sum+=$5; count++ } END {print sum/count}');
mus_theta=$(cat $i/output/pixy_watterson_theta.txt | awk '$1 == "museum" { sum+=$5; count++ } END {print sum/count}');
fold_change=$(echo "$mus_theta / $mod_theta" | bc -l);
echo -e "${i}\t${mod_theta}\t${mus_theta}\t${fold_change}"; 
done

for i in */; do
```
