#!/bin/bash

data_dir=~florat/velocity/aricia_agestis/modern_data/00_raw_reads_modc/Samples
output_tsv="C3modc_samples.tsv"
filetag=".veladapt.clean_R1.fastq.gz"

# Clear or create the output TSV file
> "$output_tsv"

# Find all *_R1.fastq.gz files and generate matching entries
for r1 in "${data_dir}"/*"${filetag}"; do
        # Get the base filename (no directory)
        r1_file=$(basename "$r1")
        sample_label=$(echo "$r1_file" | sed "s/$filetag//")
        # Derive the R2 filename from R1
        r2_tag=$(echo $filetag | sed 's/R1/R2/')
        r2_file="${sample_label}${r2_tag}"
        # Add paths back
        r1_path="${data_dir}"/"${r1_file}"
        r2_path="${data_dir}"/"${r2_file}"
        # Write line to tsv: project name, R1 file, R2 file
        echo -e "${sample_label}\t${r1_path}\t${r2_path}" >> "$output_tsv"
done

# Make a config file first called C3_config.txt

#!/bin/bash

input_tsv=C3modc_samples.tsv
output=NOVOPlasty

mkdir -p $output

while IFS=$'\t' read -r project_name forward_read reverse_read; do
    config_file="config_${project_name}.txt"
    cp C3_config.txt "${output}"/"$config_file"
    sed -i "s|^Project name.*|Project name          = ${project_name}|" "${output}"/"$config_file"
    sed -i "s|^Forward reads.*|Forward reads         = ${forward_read}|" "${output}"/"$config_file"
    sed -i "s|^Reverse reads.*|Reverse reads         = ${reverse_read}|" "${output}"/"$config_file"
    perl NOVOPlasty/NOVOPlasty4.3.5.pl -c "${output}"/"$config_file"
done < "$input_tsv"
