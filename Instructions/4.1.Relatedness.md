#!/bin/bash

# Set your directory containing the FASTQ files
data_dir="./your_data_dir"
output_tsv="samples.tsv"

# Clear or create the output TSV file
> "$output_tsv"

# Find all *_R1.fastq.gz files and generate matching entries
for r1 in "$data_dir"/*_R1.fastq.gz; do
    # Get the base filename (no directory)
    r1_file=$(basename "$r1")

    # Derive the R2 filename from R1
    r2_file="${r1_file/_R1.fastq.gz/_R2.fastq.gz}"

    # Extract sample label, e.g., SAMPLE_1 -> Sample_1
    sample_label=$(echo "$r1_file" | sed -E 's/^([A-Za-z0-9]+_[0-9]+)_FILTERED_R1.fastq.gz/Sample_\2/;s/SAMPLE/Sample/')

    # Write line to TSV: project name, R1 file, R2 file
    echo -e "${sample_label}\t${r1_file}\t${r2_file}" >> "$output_tsv"
done
